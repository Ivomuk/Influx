# PURPOSE: Quantify the commercial and risk impact of the six
# proposed BankTech loan eligibility rules on revenue, customers,
# loan value, and loan volume.
#
# SOURCES: vw1 / vw2 / vw3 / wamujap.fm_momo_loan_portfolio_cummulation.
# vw4 (scores), vw5 (reason codes), vw6 (decisions) are
# deliberately excluded as those views are not yet verified.
#
# INPUT:   eligibility_df — loaded from vw_loan_eligibility_waterfall
# OUTPUTS: eligibility waterfall | active borrowers vs proposed |
#          standalone rule impact | sensitivity scenarios |
#          risk segmentation | executive summary
#
# DATA LIMITATION NOTES:
#   Rule 1: Connected to devdata.account_holder_dump (registration_date / activation_date,
#     both VARCHAR YYYYMMDD). Falls back to first_activity_dt from vw1 when both are NULL.
#   Rules 3 & 4: 30-day proxy for 90-day rule. Set window_scale_factor=3.0
#     in apply_eligibility_rules() when full 90-day history is available.
#   Rule 5: Derived from fm_momo_loan_portfolio_cummulation (loan_repaid=0).
#     Assumes 30-day loan term; no due_date column in source table.
#     tbl_dt=20260331 — update to 20260430 when that snapshot is available.
#   Rule 6: Derived from fm_momo_loan_portfolio_cummulation (loan_repaid=1).
#     Late repayment proxy: days from disbursement to repayment > 30 days.
#     Same 30-day term assumption and tbl_dt caveat as Rule 5.
#   Revenue: loan_disb_amt_30d × base_margin_rate. No repayment probability
#     multiplier — scores from vw4 are not yet verified.
# ============================================================

def validate_eligibility_df(df):
    missing = [c for c in REQUIRED_ELIGIBILITY_COLS if c not in df.columns]
    if missing:
        raise ValueError('eligibility_df is missing columns: ' + ', '.join(missing))

def apply_eligibility_rules(df, scenario_cfg, window_scale_factor=1.0):
    """
    Apply all six eligibility rules per scenario thresholds.

    window_scale_factor: multiply 30-day txn counts/values to approximate
    the 90-day window required by Rules 3 & 4. Use 3.0 when 30-day data
    is the only available proxy for a 90-day rule.
    """
    out = df.copy()

    # Rule 1: MoMo age >= 180 days — placeholder until registry data available
    if 'rule1_momo_age_180d_flag' in out.columns:
        out['rule1_flag'] = out['rule1_momo_age_180d_flag']
    else:
        out['rule1_flag'] = 1

    # Rule 2: MAU30
    if scenario_cfg.get('rule2_required', True):
        out['rule2_flag'] = (out['active_days_last30d'] >= 1).astype(int)
    else:
        out['rule2_flag'] = 1

    # Rule 3: >= N MoMo transactions in 90 days
    out['rule3_flag'] = (
        out['momo_txn_cnt_30d'] * window_scale_factor >= scenario_cfg['rule3_min_txns']
    ).astype(int)

    # Rule 4: >= UGX X in inflows or outflows in 90 days
    out['rule4_flag'] = (
        (out['momo_inflow_30d']  * window_scale_factor >= scenario_cfg['rule4_min_value']) |
        (out['momo_outflow_30d'] * window_scale_factor >= scenario_cfg['rule4_min_value'])
    ).astype(int)

    # Rule 5: No active term loan overdue > 60 days.
    # Preferred: use pre-computed flag from the loan portfolio table (SQL view).
    # Fallback: vw3 proxy if loan portfolio columns are not present.
    if 'rule5_no_overdue_60d_flag' in out.columns:
        out['rule5_flag'] = out['rule5_no_overdue_60d_flag'].fillna(1).astype(int)
    else:
        overdue = (
            (out['outstanding_exposure_amt'] > 0) &
            (out['days_since_last_repayment'].fillna(0) > scenario_cfg['rule5_overdue_days']) &
            (out['repayment_ratio_30d'].fillna(0) < 0.10)
        )
        out['rule5_flag'] = (~overdue).astype(int)

    # Rule 6: Not in cooling-off period.
    # Preferred: use pre-computed flag from the loan portfolio table (SQL view).
    # Fallback: vw3 proxy if loan portfolio columns are not present.
    if 'rule6_no_cooling_off_flag' in out.columns:
        out['rule6_flag'] = out['rule6_no_cooling_off_flag'].fillna(1).astype(int)
    else:
        cooling = (
            (out['outstanding_exposure_amt'] > 0) &
            (out['days_since_last_repayment'].fillna(0) > scenario_cfg['rule6_repay_delay']) &
            (out['repayment_ratio_30d'].fillna(0) < scenario_cfg['rule6_ratio_cap'])
        )
        out['rule6_flag'] = (~cooling).astype(int)

    # Cumulative waterfall flags
    out['eligible_after_rule1'] = out['rule1_flag']
    out['eligible_after_rule2'] = (out['eligible_after_rule1'] & out['rule2_flag']).astype(int)
    out['eligible_after_rule3'] = (out['eligible_after_rule2'] & out['rule3_flag']).astype(int)
    out['eligible_after_rule4'] = (out['eligible_after_rule3'] & out['rule4_flag']).astype(int)
    out['eligible_after_rule5'] = (out['eligible_after_rule4'] & out['rule5_flag']).astype(int)
    out['proposed_eligible']    = (out['eligible_after_rule5'] & out['rule6_flag']).astype(int)

    return out


def add_revenue_estimate(df, config):
    """
    Estimate monthly BankTech revenue from actual disbursement amounts.
    Formula: loan_disb_amt_30d * base_margin_rate
    No repayment probability multiplier — scores from vw4 are not verified.
    """
    out = df.copy()
    out['estimated_monthly_revenue'] = out['loan_disb_amt_30d'] * config['base_margin_rate']
    return out

def build_eligibility_waterfall(df):
    """Rule-by-rule eligibility waterfall showing customer drop at each step."""
    total = len(df)
    steps = [
        ('Base',   'All MoMo subscribers',               total,                               None),
        ('Rule 1', ELIGIBILITY_RULES['rule1'],             df['eligible_after_rule1'].sum(),    None),
        ('Rule 2', ELIGIBILITY_RULES['rule2'],             df['eligible_after_rule2'].sum(),    None),
        ('Rule 3', ELIGIBILITY_RULES['rule3'],             df['eligible_after_rule3'].sum(),    None),
        ('Rule 4', ELIGIBILITY_RULES['rule4'],             df['eligible_after_rule4'].sum(),    None),
        ('Rule 5', ELIGIBILITY_RULES['rule5'],             df['eligible_after_rule5'].sum(),    None),
        ('Rule 6', ELIGIBILITY_RULES['rule6'],             df['proposed_eligible'].sum(),       None),
    ]

    rows = []
    prev = total
    for step, desc, remaining, _ in steps:
        remaining = int(remaining)
        dropped   = prev - remaining if step != 'Base' else 0
        drop_pct  = f'{dropped / prev * 100:.1f}%'  if prev > 0 and step != 'Base' else '–'
        pass_pct  = f'{remaining / total * 100:.1f}%' if total > 0 else '–'
        rows.append({
            'Step':                step,
            'Rule Applied':        desc,
            'Customers Remaining': f'{remaining:,}',
            'Customers Dropped':   f'{dropped:,}' if step != 'Base' else '–',
            'Drop %':              drop_pct,
            'Cumulative Pass %':   pass_pct,
        })
        prev = remaining

    return pd.DataFrame(rows)


def build_impact_comparison(df):
    """
    Currently active borrowers vs proposed eligible subscribers.

    Baseline: currently_active_borrower_flag = 1 (had disbursement in window).
    This replaces the "currently_eligible" view from vw6 which is not yet verified.
    Risk quality is measured via raw repayment and activity metrics — no DSI or
    fraud scores.
    """
    active   = df[df['currently_active_borrower_flag'] == 1]
    proposed = df[df['proposed_eligible'] == 1]

    def fmt(v):
        return f'{v:,.0f}' if pd.notna(v) else '–'

    def pct_chg(new, old):
        if old == 0 or pd.isna(old):
            return 'N/A'
        return f'{(new - old) / old * 100:+.1f}%'

    def diff(new, old):
        if pd.isna(new) or pd.isna(old):
            return '–'
        return f'{new - old:+,.0f}'

    rows = [
        ('Customers',
         len(active), len(proposed)),
        ('Loan disbursements (count, 30d)',
         int(active['disbursement_cnt_30d'].sum()),   int(proposed['disbursement_cnt_30d'].sum())),
        ('Total loan disbursement value (UGX)',
         active['loan_disb_amt_30d'].sum(),            proposed['loan_disb_amt_30d'].sum()),
        ('Total loan repaid value (UGX)',
         active['loan_repaid_amt_30d'].sum(),          proposed['loan_repaid_amt_30d'].sum()),
        ('Outstanding exposure (UGX)',
         active['outstanding_exposure_amt'].sum(),     proposed['outstanding_exposure_amt'].sum()),
        ('Total wallet inflow (UGX)',
         active['wallet_inflow_amt_30d'].sum(),        proposed['wallet_inflow_amt_30d'].sum()),
        ('Total wallet outflow (UGX)',
         active['wallet_outflow_amt_30d'].sum(),       proposed['wallet_outflow_amt_30d'].sum()),
        ('Est. monthly BankTech revenue (UGX)',
         active['estimated_monthly_revenue'].sum(),    proposed['estimated_monthly_revenue'].sum()),
        ('Avg repayment ratio',
         active['repayment_ratio_30d'].mean(),         proposed['repayment_ratio_30d'].mean()),
        ('Avg days since last repayment',
         active['days_since_last_repayment'].mean(),   proposed['days_since_last_repayment'].mean()),
        ('Avg outstanding exposure per borrower (UGX)',
         active['outstanding_exposure_amt'].mean(),    proposed['outstanding_exposure_amt'].mean()),
        ('Stacked borrowing rate %',
         active['stacked_borrowing_flag_30d'].mean() * 100,
         proposed['stacked_borrowing_flag_30d'].mean() * 100),
        ('Avg wallet inflow per subscriber (UGX)',
         active['wallet_inflow_amt_30d'].mean(),       proposed['wallet_inflow_amt_30d'].mean()),
    ]

    output = []
    for metric, c_val, p_val in rows:
        output.append({
            'Metric':                       metric,
            'Active Borrowers (baseline)':  fmt(c_val),
            'Proposed Eligible':            fmt(p_val),
            'Change':                       diff(p_val, c_val),
            'Change %':                     pct_chg(p_val, c_val),
        })

    return pd.DataFrame(output)


def build_standalone_rule_impact(df):
    """
    Each rule tested in isolation: how many customers it would exclude
    on its own, and the loan value / revenue associated with those customers.
    """
    total = len(df)
    rule_flag_map = [
        ('Rule 1', ELIGIBILITY_RULES['rule1'], 'rule1_flag'),
        ('Rule 2', ELIGIBILITY_RULES['rule2'], 'rule2_flag'),
        ('Rule 3', ELIGIBILITY_RULES['rule3'], 'rule3_flag'),
        ('Rule 4', ELIGIBILITY_RULES['rule4'], 'rule4_flag'),
        ('Rule 5', ELIGIBILITY_RULES['rule5'], 'rule5_flag'),
        ('Rule 6', ELIGIBILITY_RULES['rule6'], 'rule6_flag'),
    ]

    rows = []
    for rule_id, rule_desc, flag_col in rule_flag_map:
        fail_df = df[df[flag_col] == 0]
        pass_df = df[df[flag_col] == 1]
        fail_n  = len(fail_df)

        rows.append({
            'Rule':                            rule_id,
            'Description':                     rule_desc,
            'Customers Failing':               f'{fail_n:,}',
            'Fail %':                          f'{fail_n / total * 100:.1f}%' if total > 0 else '–',
            'Loan Value at Risk (UGX)':        f'{fail_df["loan_disb_amt_30d"].sum():,.0f}',
            'Revenue at Risk (UGX)':           f'{fail_df["estimated_monthly_revenue"].sum():,.0f}',
            'Avg Repayment Ratio (failing)':   f'{fail_df["repayment_ratio_30d"].mean():.2f}' if fail_n > 0 else '–',
            'Avg Days Since Repayment (fail)': f'{fail_df["days_since_last_repayment"].mean():.0f}' if fail_n > 0 else '–',
            'Avg Repayment Ratio (passing)':   f'{pass_df["repayment_ratio_30d"].mean():.2f}' if len(pass_df) > 0 else '–',
        })

    return pd.DataFrame(rows)


def build_scenario_comparison(df_base, scenarios, config):
    """
    Compare eligible population and key metrics across the three
    threshold scenarios (proposed / less strict / more strict).
    df_base: raw eligibility DataFrame before rules are applied.
    """
    rows = []
    for _, scenario_cfg in scenarios.items():
        s_df     = apply_eligibility_rules(
            df_base, scenario_cfg,
            window_scale_factor=config.get('window_scale_factor', 1.0)
        )
        s_df     = add_revenue_estimate(s_df, config)
        eligible = s_df[s_df['proposed_eligible'] == 1]
        n_elig   = len(eligible)
        total    = len(s_df)

        rows.append({
            'Scenario':                  scenario_cfg['label'],
            'Rule 3 Min Txns':           scenario_cfg['rule3_min_txns'],
            'Rule 4 Min Value (UGX)':    f"{scenario_cfg['rule4_min_value']:,}",
            'Rule 5 Overdue Days':       scenario_cfg['rule5_overdue_days'],
            'Rule 6 Repay Delay (days)': scenario_cfg['rule6_repay_delay'],
            'Eligible Customers':        f'{n_elig:,}',
            'Eligible %':                f'{n_elig / total * 100:.1f}%' if total > 0 else '–',
            'Total Loan Value (UGX)':    f'{eligible["loan_disb_amt_30d"].sum():,.0f}',
            'Est. Revenue (UGX)':        f'{eligible["estimated_monthly_revenue"].sum():,.0f}',
            'Avg Repayment Ratio':       f'{eligible["repayment_ratio_30d"].mean():.2f}' if n_elig > 0 else '–',
            'Avg Days Since Repayment':  f'{eligible["days_since_last_repayment"].mean():.0f}' if n_elig > 0 else '–',
        })

    return pd.DataFrame(rows)


def build_risk_segmentation(df):
    """
    Compare pass-all-rules vs fail-one-or-more segments on financial
    and repayment-behaviour metrics. No score-derived fields used.
    """
    segments = {
        'Pass all rules':         df[df['proposed_eligible'] == 1],
        'Fail one or more rules': df[df['proposed_eligible'] == 0],
    }

    rows = []
    for label, seg in segments.items():
        n         = len(seg)
        borrowers = seg[seg['loan_disb_amt_30d'] > 0]
        if n == 0:
            continue
        rows.append({
            'Segment':                                label,
            'Customers':                              f'{n:,}',
            'Loan Count (30d disbursements)':         f'{int(seg["disbursement_cnt_30d"].sum()):,}',
            'Total Loan Value (UGX)':                 f'{seg["loan_disb_amt_30d"].sum():,.0f}',
            'Avg Loan Size (UGX)':                    f'{borrowers["loan_disb_amt_30d"].mean():,.0f}' if len(borrowers) > 0 else '0',
            'Est. Monthly Revenue (UGX)':             f'{seg["estimated_monthly_revenue"].sum():,.0f}',
            'Avg Repayment Ratio':                    f'{seg["repayment_ratio_30d"].mean():.2f}' if seg['repayment_ratio_30d'].notna().any() else '–',
            'Avg Outstanding Exposure (UGX)':         f'{seg["outstanding_exposure_amt"].mean():,.0f}',
            'Avg Days Since Last Repayment':          f'{seg["days_since_last_repayment"].mean():.0f}' if seg['days_since_last_repayment'].notna().any() else '–',
            'Stacked Borrowing Rate %':               f'{seg["stacked_borrowing_flag_30d"].mean() * 100:.1f}%',
            'Repeated Borrowing Rate %':              f'{seg["repeated_borrowing_flag_30d"].mean() * 100:.1f}%',
            'Avg Wallet Inflow (UGX)':                f'{seg["wallet_inflow_amt_30d"].mean():,.0f}',
            # Loan portfolio signals
            'Customers with overdue loans':           f'{int((seg["overdue_loan_count"] > 0).sum()):,}',
            'Total overdue loan exposure (UGX)':      f'{seg["total_overdue_amount"].sum():,.0f}',
            'Customers in cooling-off':               f'{int((seg["cooling_off_loan_count"] > 0).sum()):,}',
            'Cooling-off affected loan value (UGX)':  f'{seg["cooling_off_affected_loan_value"].sum():,.0f}',
        })

    return pd.DataFrame(rows)


def build_executive_summary(df, waterfall_df, config):
    """One-page executive summary using only verified vw1/vw2/vw3 signals."""
    total          = len(df)
    n_active       = int(df['currently_active_borrower_flag'].sum())
    n_proposed     = int(df['proposed_eligible'].sum())
    net_change     = n_proposed - n_active
    net_change_pct = net_change / n_active * 100 if n_active > 0 else 0

    rev_active     = df[df['currently_active_borrower_flag'] == 1]['estimated_monthly_revenue'].sum()
    rev_proposed   = df[df['proposed_eligible'] == 1]['estimated_monthly_revenue'].sum()
    rev_change     = rev_proposed - rev_active
    rev_change_pct = rev_change / rev_active * 100 if rev_active > 0 else 0

    val_active     = df[df['currently_active_borrower_flag'] == 1]['loan_disb_amt_30d'].sum()
    val_proposed   = df[df['proposed_eligible'] == 1]['loan_disb_amt_30d'].sum()
    val_change_pct = (val_proposed - val_active) / val_active * 100 if val_active > 0 else 0

    rep_active     = df[df['currently_active_borrower_flag'] == 1]['repayment_ratio_30d'].mean()
    rep_proposed   = df[df['proposed_eligible'] == 1]['repayment_ratio_30d'].mean()

    exp_active     = df[df['currently_active_borrower_flag'] == 1]['outstanding_exposure_amt'].mean()
    exp_proposed   = df[df['proposed_eligible'] == 1]['outstanding_exposure_amt'].mean()

    # Most restrictive rule: largest absolute drop in the waterfall
    wf = waterfall_df[waterfall_df['Step'] != 'Base'].copy()
    wf['_dropped_num'] = (
        wf['Customers Dropped']
        .str.replace(',', '', regex=False)
        .replace('–', '0')
        .apply(lambda x: int(x) if str(x).lstrip('+-').isdigit() else 0)
    )
    most_restrictive = wf.loc[wf['_dropped_num'].idxmax(), 'Step'] if len(wf) > 0 else 'N/A'

    # Rule 1 exclusion count (now live from account_holder_dump)
    r1_excluded = int((df['rule1_momo_age_180d_flag'] == 0).sum()) \
        if 'rule1_momo_age_180d_flag' in df.columns else None
    r1_status = (
        f'Active — account_holder_dump connected ({r1_excluded:,} excluded)'
        if r1_excluded is not None
        else 'Pending — rule1_momo_age_180d_flag not in dataset'
    )

    metrics = [
        ('Total addressable subscriber base',           f'{total:,}'),
        ('Currently active borrowers (baseline)',       f'{n_active:,}'),
        ('Proposed eligible (new rules)',               f'{n_proposed:,}'),
        ('Net change vs active borrowers',              f'{net_change:+,} ({net_change_pct:+.1f}%)'),
        ('',                                            ''),
        ('Est. monthly revenue – active borrowers (UGX)', f'{rev_active:,.0f}'),
        ('Est. monthly revenue – proposed (UGX)',        f'{rev_proposed:,.0f}'),
        ('Revenue change (UGX)',                        f'{rev_change:+,.0f} ({rev_change_pct:+.1f}%)'),
        (' ',                                           ''),
        ('Loan disbursement value – active (UGX)',      f'{val_active:,.0f}'),
        ('Loan disbursement value – proposed (UGX)',    f'{val_proposed:,.0f}'),
        ('Loan value change %',                         f'{val_change_pct:+.1f}%'),
        ('  ',                                          ''),
        ('Avg repayment ratio – active borrowers',      f'{rep_active:.2f}' if pd.notna(rep_active) else '–'),
        ('Avg repayment ratio – proposed eligible',     f'{rep_proposed:.2f}' if pd.notna(rep_proposed) else '–'),
        ('Avg outstanding exposure – active (UGX)',     f'{exp_active:,.0f}' if pd.notna(exp_active) else '–'),
        ('Avg outstanding exposure – proposed (UGX)',   f'{exp_proposed:,.0f}' if pd.notna(exp_proposed) else '–'),
        ('   ',                                         ''),
        ('Most restrictive rule',                       most_restrictive),
        ('   ',                                         ''),
        # Rule 1 exclusions (from devdata.account_holder_dump)
        ('Customers excluded by Rule 1 (MoMo age < 180 days)',
         f'{r1_excluded:,}' if r1_excluded is not None else 'Pending'),
        # Loan portfolio summary (from fm_momo_loan_portfolio_cummulation)
        ('Customers excluded by Rule 5 (overdue)',
         f'{int((df["overdue_loan_count"] > 0).sum()):,}'),
        ('Total overdue exposure removed (UGX)',
         f'{df["total_overdue_amount"].sum():,.0f}'),
        ('Customers excluded by Rule 6 (cooling-off)',
         f'{int((df["cooling_off_loan_count"] > 0).sum()):,}'),
        ('Cooling-off affected loan value (UGX)',
         f'{df["cooling_off_affected_loan_value"].sum():,.0f}'),
        ('    ',                                        ''),
        ('Data window',                                 f'{config["data_window_days"]} days (April 2026)'),
        ('Sources used',                                'vw1, vw2, vw3, account_holder_dump, fm_momo_loan_portfolio_cummulation'),
        ('Revenue method',                              'loan_disb_amt_30d × 10% margin (no score multiplier)'),
        ('Rule 1 status',                               r1_status),
        ('Rules 3 & 4 status',                         '30-day proxy for 90-day window (extend window_90d_start in params CTE)'),
        ('Rules 5 & 6 status',                         'Loan portfolio table — 30-day loan term confirmed'),
    ]

    return pd.DataFrame(metrics, columns=['Metric', 'Value'])


# ============================================================
# MAIN RUNNER
# ============================================================

def run_loan_eligibility_impact_analysis(eligibility_df, config=None, scenarios=None):
    """
    Full loan eligibility impact analysis pipeline.

    Parameters
    ----------
    eligibility_df : pd.DataFrame
        Output from vw_loan_eligibility_waterfall SQL view.
        Must contain all columns in REQUIRED_ELIGIBILITY_COLS.
    config : dict, optional
        Overrides ANALYSIS_CONFIG defaults.
    scenarios : dict, optional
        Overrides SCENARIO_CONFIGS for sensitivity analysis.

    Returns
    -------
    dict with keys:
        executive_summary        : pd.DataFrame
        eligibility_waterfall    : pd.DataFrame
        impact_comparison        : pd.DataFrame   (active borrowers vs proposed)
        standalone_rule_impact   : pd.DataFrame
        scenario_comparison      : pd.DataFrame
        risk_segmentation        : pd.DataFrame
        proposed_eligible_df     : pd.DataFrame
        ineligible_df            : pd.DataFrame
        full_analysis_df         : pd.DataFrame
    """
    local_config    = {**ANALYSIS_CONFIG,   **(config    or {})}
    local_scenarios = {**SCENARIO_CONFIGS,  **(scenarios or {})}

    validate_eligibility_df(eligibility_df)

    df = eligibility_df.copy()
    numeric_cols = [c for c in REQUIRED_ELIGIBILITY_COLS if c != 'subscriber_msisdn']
    for col in numeric_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')

    # Apply proposed rules and compute revenue
    df = apply_eligibility_rules(
        df,
        local_scenarios['proposed'],
        window_scale_factor=local_config.get('window_scale_factor', 1.0),
    )
    df = add_revenue_estimate(df, local_config)

    # Build analysis tables
    waterfall_df  = build_eligibility_waterfall(df)
    impact_df     = build_impact_comparison(df)
    standalone_df = build_standalone_rule_impact(df)
    scenario_df   = build_scenario_comparison(eligibility_df.copy(), local_scenarios, local_config)
    risk_df       = build_risk_segmentation(df)
    summary_df    = build_executive_summary(df, waterfall_df, local_config)

    # ── Print outputs ──────────────────────────────────────────
    sep = '=' * 80
    print(sep)
    print('  LOAN ELIGIBILITY IMPACT ANALYSIS — BANKTECH')
    print(f'  Data window: April 2026 (30d)  |  Version: {local_config["output_version"]}')
    print('  Sources: vw1, vw2, vw3, fm_momo_loan_portfolio_cummulation. Scores/decisions excluded.')
    print(sep)

    print('\n── EXECUTIVE SUMMARY ──')
    print(summary_df.to_string(index=False))

    print('\n── ELIGIBILITY WATERFALL ──')
    print(waterfall_df.to_string(index=False))

    print('\n── ACTIVE BORROWERS vs PROPOSED ELIGIBLE — IMPACT COMPARISON ──')
    print(impact_df.to_string(index=False))

    print('\n── STANDALONE RULE IMPACT (each rule tested independently) ──')
    print(standalone_df.to_string(index=False))

    print('\n── SENSITIVITY SCENARIOS ──')
    print(scenario_df.to_string(index=False))

    print('\n── RISK SEGMENTATION: PASS vs FAIL ──')
    print(risk_df.to_string(index=False))

    return {
        'executive_summary':      summary_df,
        'eligibility_waterfall':  waterfall_df,
        'impact_comparison':      impact_df,
        'standalone_rule_impact': standalone_df,
        'scenario_comparison':    scenario_df,
        'risk_segmentation':      risk_df,
        'proposed_eligible_df':   df[df['proposed_eligible'] == 1].copy(),
        'ineligible_df':          df[df['proposed_eligible'] == 0].copy(),
        'full_analysis_df':       df,
    }
