insert into {TARGET_MART_TABLE}

-- ─────────────────────────────────────────────────────────────────────────────
-- Changes from original (subs_eg_original.txt):
--
-- Rule 1 (momo_age_180, was mau_180):
--   Renamed CTE mau_180 → momo_age_180.
--   Source changed from analytics.momo_rge_base_subscribers_v9 (proxy: was
--   subscriber active 179 days ago?) to devdata.account_holder_dump (actual
--   registration_date / activation_date). Parses registration_date first,
--   falls back to activation_date, then checks date_diff >= 180 days.
--
-- Rule 5 (active_overdue_60d):
--   Source changed from wamujap.fm_momo_loan_portfolio_cummulation
--   (loan_repaid = 0, manual date diff on disbursement_date) to
--   analytics.momo_loan_book_tracker_loan_state_daily (pre-computed
--   outstanding_ugx > 0 and days_aging > 60).
--
-- Rule 6 (cooling_off):
--   Source changed from wamujap.fm_momo_loan_portfolio_cummulation
--   (loan_repaid = 1, repayment_date, disbursement_date) to
--   analytics.momo_loan_book_tracker_loan_state_daily.
--   Scans 90-day window of partitions (closed loans appear on repayment day
--   then drop from subsequent partitions). days_aging is NULL for settled
--   loans so lateness is derived from: date_diff(due_date, last_repayment_date)
--   > 30, where due_date = last_disbursement_date + tenure_days.
--   MOMO-ADVANCE loans excluded via tenure_days IS NOT NULL (they are same-day
--   micro-loans with no tenure and are not subject to cooling-off).
--
-- eligible CTE:
--   Join updated from mau_180 → momo_age_180 to match renamed CTE.
--
-- Unchanged: raw_txns, sub_activity, mau_30, derisk_exclusions, final SELECT.
-- ─────────────────────────────────────────────────────────────────────────────

with

-- ─────────────────────────────────────────────────────────────────────────────
-- Rules 3 & 4: 90-day transaction activity from fin_log_cfw
-- Outbound = subscriber as sender; Inbound = subscriber as receiver.
-- Only subscriber profiles are counted (excludes agents, merchants, etc.).
-- ─────────────────────────────────────────────────────────────────────────────
raw_txns as (
    select
        instruct_hdr_fid,
        instruct_from_fro_msisdn    as msisdn,
        instruct_amount,
        'OUT'                       as direction
    from feeds.fin_log_cfw
    where tbl_dt between cast(date_format(date_add('day', -89, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
                     and {RUN_DATE_KEY}
      and status_code = 'EXECUTED'
      and hdr_type = 'RESERVATION'
      and instruct_from_fro_msisdn is not null
      and instruct_from_fro_user_prf in (
            select profile_name from analytics.mobile_money_profiles where subscriber_flag = true
          )

    union all

    select
        instruct_hdr_fid,
        instruct_to_fro_msisdn      as msisdn,
        instruct_amount,
        'IN'                        as direction
    from feeds.fin_log_cfw
    where tbl_dt between cast(date_format(date_add('day', -89, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
                     and {RUN_DATE_KEY}
      and status_code = 'EXECUTED'
      and hdr_type = 'RESERVATION'
      and instruct_to_fro_msisdn is not null
      and instruct_to_fro_user_prf in (
            select profile_name from analytics.mobile_money_profiles where subscriber_flag = true
          )
),

sub_activity as (
    select
        msisdn,
        count(distinct instruct_hdr_fid)                                    as txn_count,
        sum(case when direction = 'IN'  then instruct_amount else 0 end)    as inflow_amt,
        sum(case when direction = 'OUT' then instruct_amount else 0 end)    as outflow_amt
    from raw_txns
    group by msisdn
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 2: Customer must be MAU30 on the run date.
-- ─────────────────────────────────────────────────────────────────────────────
mau_30 as (
    select distinct msisdn
    from analytics.momo_rge_base_subscribers_v9
    where date_key = {RUN_DATE_KEY}
      and au_30_day_active_flag = 1
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 1: Age on MoMo >= 180 days.
-- Source: devdata.account_holder_dump (MOBILE MONEY, ACTIVE subscribers only).
-- Derivation: registration_date first, activation_date as fallback.
-- Both columns are VARCHAR DD-MON-YY ('18-NOV-25') — parsed via date_parse.
-- ─────────────────────────────────────────────────────────────────────────────
momo_age_180 as (
    select msisdn
    from (
        select
            msisdn,
            COALESCE(
                TRY(date_parse(registration_date, '%d-%b-%y')),
                TRY(date_parse(activation_date,   '%d-%b-%y'))
            ) as derived_reg_date
        from devdata.account_holder_dump
        where tbl_dt               = {RUN_DATE_KEY}
          and account_type         = 'MOBILE MONEY'
          and accountholder_status = 'ACTIVE'
          and account_status       = 'ACTIVE'
          and profile in (select profile_name from analytics.mobile_money_profiles where subscriber_flag = true)
    )
    where derived_reg_date is not null
      and date_diff('day', derived_reg_date, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')) >= 180
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Derisk exclusion: MTNU HV Gold Subscriber and MTNU Agent Derisk Class
-- profiles are systematically excluded from lending partner distribution.
-- ─────────────────────────────────────────────────────────────────────────────
derisk_exclusions as (
    select distinct msisdn
    from devdata.account_holder_dump
    where tbl_dt = {RUN_DATE_KEY}
      and profile in ('MTNU Agent Derisk Class', 'MTNU HV Gold Subscriber')
   
 -- Double check if this rule is VALID
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 5: No active term loan overdue by more than 60 days.
-- days_aging is pre-computed in the source table — no date arithmetic needed.
-- ─────────────────────────────────────────────────────────────────────────────
active_overdue_60d as (
    select distinct customer_msisdn as msisdn
    from analytics.momo_loan_book_tracker_loan_state_daily
    where date_key      = {RUN_DATE_KEY}
      and outstanding_ugx > 0
      and days_aging    > 60
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 6: 90-day cooling-off after a delayed loan repayment.
-- "Delayed" = principal repaid with days_past_due > 30.
-- days_past_due is frozen at the value when repayment was made.
-- Cooling-off runs 90 days from closure_date (repayment date).
-- Source: xtrafloat table retains settled loan history; loan_state_daily does not.
-- ─────────────────────────────────────────────────────────────────────────────
cooling_off as (
    select distinct customer_msisdn as msisdn
    from analytics.momo_loan_book_tracker_loan_state_daily
    where date_key between cast(date_format(date_add('day', -90, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
                      and {RUN_DATE_KEY}
      and loan_status         in ('CLOSED', 'SETTLED', 'OVERPAID')
      and tenure_days         is not null
      and last_repayment_date is not null
      and date_diff('day',
            date_add('day', tenure_days, last_disbursement_date),
            last_repayment_date
          ) > 30
      and date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')
          between last_repayment_date
              and date_add('day', 90, last_repayment_date)
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Final eligibility gate: all 6 rules must pass.
-- Rules 1 & 2 → inner joins on mau_180 and mau_30.
-- Rules 3 & 4 → txn_count >= 5 AND (inflows + outflows) >= UGX 10,000.
-- Rules 5 & 6 → NOT IN overdue / cooling-off sets.
-- Derisk profiles are also excluded.
-- ─────────────────────────────────────────────────────────────────────────────
eligible as (
    select
        m30.msisdn,
        coalesce(a.txn_count,   0)  as txn_count_90d,
        coalesce(a.inflow_amt,  0)  as inflow_amt_90d,
        coalesce(a.outflow_amt, 0)  as outflow_amt_90d
    from mau_30 m30
    inner join momo_age_180 m180
        on m30.msisdn = m180.msisdn
    inner join sub_activity a
        on m30.msisdn = a.msisdn
    where a.txn_count >= 5
      and (a.inflow_amt + a.outflow_amt) >= 10000
      and m30.msisdn not in (select msisdn from derisk_exclusions)
      and m30.msisdn not in (select msisdn from active_overdue_60d)
      and m30.msisdn not in (select msisdn from cooling_off)
)

select
    msisdn,
    txn_count_90d,
    inflow_amt_90d,
    outflow_amt_90d,
    cast(1 as tinyint)      as eligible_flag,
    current_timestamp       as inserted_dt,
    {RUN_DATE_KEY}          as tbl_dt
from eligible
