CREATE OR REPLACE VIEW vw3_credit_v1_layer1_features AS
WITH anchor_days AS (
    SELECT DISTINCT
        event_dt AS feature_dt,
        subscriber_msisdn
    FROM vw2_credit_v1_subscriber_day
),
window_join AS (
    SELECT
        a.feature_dt,
        a.subscriber_msisdn,
        b.event_dt,
        b.loan_disb_amt_day,
        b.loan_repaid_amt_day,
        b.wallet_inflow_amt_day,
        b.wallet_outflow_amt_day,
        b.spend_amt_day,
        b.loan_disb_cnt_day,
        b.loan_repay_cnt_day,
        b.wallet_txn_cnt_day,
        b.lender_family_cnt_day,
        b.had_disbursement_day,
        b.had_repayment_day,
        b.alt_credit_signal_flag_day
    FROM anchor_days a
    INNER JOIN vw2_credit_v1_subscriber_day b
        ON a.subscriber_msisdn = b.subscriber_msisdn
       AND b.event_dt BETWEEN date_add('day', -29, a.feature_dt) AND a.feature_dt
),
window_agg AS (
    SELECT
        feature_dt,
        subscriber_msisdn,

        SUM(loan_disb_amt_day) AS loan_disb_amt_30d,
        SUM(loan_repaid_amt_day) AS loan_repaid_amt_30d,
        SUM(wallet_inflow_amt_day) AS wallet_inflow_amt_30d,
        SUM(wallet_outflow_amt_day) AS wallet_outflow_amt_30d,
        SUM(spend_amt_day) AS spend_amt_30d,

        SUM(loan_disb_cnt_day) AS disbursement_cnt_30d,
        SUM(loan_repay_cnt_day) AS repayment_cnt_30d,
        SUM(wallet_txn_cnt_day) AS wallet_txn_cnt_30d,

        COUNT(DISTINCT CASE
            WHEN lender_family_cnt_day > 0 THEN event_dt
            ELSE NULL
        END) AS active_lender_days_30d,

        MAX(lender_family_cnt_day) AS active_lender_cnt_30d,
        MAX(alt_credit_signal_flag_day) AS alt_credit_active_flag_30d,

        MAX(CASE
            WHEN had_disbursement_day = 1 THEN event_dt
            ELSE NULL
        END) AS last_disbursement_dt,

        MAX(CASE
            WHEN had_repayment_day = 1 THEN event_dt
            ELSE NULL
        END) AS last_repayment_dt,

        COUNT(DISTINCT CASE
            WHEN wallet_txn_cnt_day > 0 THEN event_dt
            ELSE NULL
        END) AS wallet_active_days_30d
    FROM window_join
    GROUP BY
        feature_dt,
        subscriber_msisdn
),
post_loan_activity AS (
    SELECT
        a.feature_dt,
        a.subscriber_msisdn,

        SUM(CASE
            WHEN b.event_dt BETWEEN date_add('day', -7, a.feature_dt) AND a.feature_dt
                THEN b.wallet_inflow_amt_day + b.wallet_outflow_amt_day + b.spend_amt_day
            ELSE 0.0
        END) AS wallet_activity_last_7d,

        SUM(CASE
            WHEN b.event_dt BETWEEN date_add('day', -14, a.feature_dt) AND date_add('day', -8, a.feature_dt)
                THEN b.wallet_inflow_amt_day + b.wallet_outflow_amt_day + b.spend_amt_day
            ELSE 0.0
        END) AS wallet_activity_prev_7d
    FROM anchor_days a
    INNER JOIN vw2_credit_v1_subscriber_day b
        ON a.subscriber_msisdn = b.subscriber_msisdn
       AND b.event_dt BETWEEN date_add('day', -14, a.feature_dt) AND a.feature_dt
    GROUP BY
        a.feature_dt,
        a.subscriber_msisdn
),
base_features AS (
    SELECT
        w.feature_dt,
        w.subscriber_msisdn,
        w.loan_disb_amt_30d,
        w.loan_repaid_amt_30d,
        w.wallet_inflow_amt_30d,
        w.wallet_outflow_amt_30d,
        w.spend_amt_30d,
        w.disbursement_cnt_30d,
        w.repayment_cnt_30d,
        w.wallet_txn_cnt_30d,
        w.active_lender_cnt_30d,
        w.active_lender_days_30d,
        w.alt_credit_active_flag_30d,
        w.last_disbursement_dt,
        w.last_repayment_dt,
        w.wallet_active_days_30d,
        p.wallet_activity_last_7d,
        p.wallet_activity_prev_7d,

        (w.loan_disb_amt_30d - w.loan_repaid_amt_30d) AS outstanding_exposure_amt,

        CASE
            WHEN w.loan_disb_amt_30d > 0 THEN w.loan_repaid_amt_30d / w.loan_disb_amt_30d
            ELSE NULL
        END AS repayment_ratio_30d,

        CASE
            WHEN w.last_disbursement_dt IS NOT NULL
                THEN date_diff('day', w.last_disbursement_dt, w.feature_dt)
            ELSE NULL
        END AS days_since_last_disbursement,

        CASE
            WHEN w.last_repayment_dt IS NOT NULL
                THEN date_diff('day', w.last_repayment_dt, w.feature_dt)
            ELSE NULL
        END AS days_since_last_repayment
    FROM window_agg w
    LEFT JOIN post_loan_activity p
        ON w.feature_dt = p.feature_dt
       AND w.subscriber_msisdn = p.subscriber_msisdn
),
final_features AS (
    SELECT
        feature_dt,
        subscriber_msisdn,
        loan_disb_amt_30d,
        loan_repaid_amt_30d,
        wallet_inflow_amt_30d,
        wallet_outflow_amt_30d,
        spend_amt_30d,
        disbursement_cnt_30d,
        repayment_cnt_30d,
        wallet_txn_cnt_30d,
        active_lender_cnt_30d,
        active_lender_days_30d,
        alt_credit_active_flag_30d,
        last_disbursement_dt,
        last_repayment_dt,
        wallet_active_days_30d,
        wallet_activity_last_7d,
        wallet_activity_prev_7d,
        outstanding_exposure_amt,
        repayment_ratio_30d,
        days_since_last_disbursement,
        days_since_last_repayment,

        CASE
            WHEN wallet_activity_prev_7d > 0
                THEN (wallet_activity_last_7d - wallet_activity_prev_7d) / wallet_activity_prev_7d
            ELSE NULL
        END AS post_loan_wallet_activity_change_ratio,

        CASE
            WHEN active_lender_cnt_30d >= 2 THEN 1
            ELSE 0
        END AS stacked_borrowing_flag_30d,

        CASE
            WHEN disbursement_cnt_30d >= 3 THEN 1
            ELSE 0
        END AS repeated_borrowing_flag_30d
    FROM base_features
)
SELECT
    feature_dt,
    subscriber_msisdn,
    outstanding_exposure_amt,
    active_lender_cnt_30d,
    disbursement_cnt_30d,
    days_since_last_disbursement,
    repayment_ratio_30d,
    repayment_cnt_30d,
    days_since_last_repayment,
    wallet_inflow_amt_30d,
    wallet_outflow_amt_30d,
    spend_amt_30d,
    wallet_txn_cnt_30d,
    wallet_active_days_30d,
    post_loan_wallet_activity_change_ratio,
    stacked_borrowing_flag_30d,
    repeated_borrowing_flag_30d,
    alt_credit_active_flag_30d,
    active_lender_days_30d,
    loan_disb_amt_30d,
    loan_repaid_amt_30d,
    last_disbursement_dt,
    last_repayment_dt,
    wallet_activity_last_7d,
    wallet_activity_prev_7d
FROM final_features
;
