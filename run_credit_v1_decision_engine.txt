# Define reusable production-grade decision-engine functions and run them on the in-memory sample data.
import pandas as pd
import numpy as np
from tqdm import tqdm

tqdm.pandas()

def validate_required_columns(input_df, required_cols, df_name):
    missing_cols = [col_name for col_name in required_cols if col_name not in input_df.columns]
    if len(missing_cols) > 0:
        raise ValueError(df_name + ' is missing required columns: ' + ', '.join(missing_cols))

def coerce_numeric_columns(input_df, numeric_cols):
    output_df = input_df.copy()
    for col_name in numeric_cols:
        output_df[col_name] = pd.to_numeric(output_df[col_name], errors='coerce')
    return output_df

def prepare_decision_base(layer1_input_df, layer0_input_df):
    validate_required_columns(layer1_input_df, REQUIRED_LAYER1_COLS, 'layer1_input_df')
    validate_required_columns(layer0_input_df, REQUIRED_LAYER0_COLS, 'layer0_input_df')
    layer1_clean_df = layer1_input_df.copy()
    layer0_clean_df = layer0_input_df.copy()
    layer1_clean_df['feature_dt'] = pd.to_datetime(layer1_clean_df['feature_dt'])
    layer0_clean_df['feature_dt'] = pd.to_datetime(layer0_clean_df['feature_dt'])
    layer1_numeric_cols = [col_name for col_name in REQUIRED_LAYER1_COLS if col_name not in ['feature_dt', 'subscriber_msisdn']]
    layer0_numeric_cols = [col_name for col_name in REQUIRED_LAYER0_COLS if col_name not in ['feature_dt', 'subscriber_msisdn']]
    layer1_clean_df = coerce_numeric_columns(layer1_clean_df, layer1_numeric_cols)
    layer0_clean_df = coerce_numeric_columns(layer0_clean_df, layer0_numeric_cols)
    merged_df = layer1_clean_df.merge(layer0_clean_df, on=['feature_dt', 'subscriber_msisdn'], how='inner', validate='one_to_one')
    if merged_df.shape[0] == 0:
        raise ValueError('Merged decision base is empty after joining layer1_input_df and layer0_input_df')
    return merged_df

def build_policy_prefilter(decision_base_input_df, config_dict):
    output_df = decision_base_input_df.copy()
    output_df['exposure_to_inflow_ratio'] = np.where(output_df['wallet_inflow_amt_30d'] > 0, output_df['outstanding_exposure_amt'] / output_df['wallet_inflow_amt_30d'], 999.0)
    output_df['thin_file_flag'] = (output_df['wallet_active_days_30d'] < config_dict['thin_file_wallet_days_threshold']).astype(int)
    output_df['recent_disbursement_no_repayment_flag'] = ((output_df['days_since_last_disbursement'] <= config_dict['recent_disb_days_threshold']) & (output_df['repayment_ratio_30d'] < config_dict['recent_disb_repayment_threshold'])).astype(int)
    output_df['policy_restrict_flag'] = ((output_df['identity_confidence_score_v1_rule'] < config_dict['identity_confidence_restrict_threshold']) | (output_df['fraud_abuse_risk_score_v1_rule'] >= config_dict['fraud_restrict_threshold']) | (output_df['debt_stress_index_v1'] >= config_dict['severe_dsi_threshold']) | (output_df['recent_disbursement_no_repayment_flag'] == 1)).astype(int)
    output_df['policy_reduce_only_flag'] = ((output_df['debt_stress_index_v1'].between(config_dict['reduce_only_dsi_min'], config_dict['reduce_only_dsi_max'])) | (output_df['repayment_ratio_30d'] < config_dict['low_repayment_ratio_threshold']) | (output_df['stacked_borrowing_flag_30d'] == 1) | (output_df['active_lender_cnt_30d'] >= config_dict['high_active_lender_cnt_threshold']) | (output_df['exposure_to_inflow_ratio'] > config_dict['high_exposure_to_inflow_threshold'])).astype(int)
    output_df['policy_maintain_only_flag'] = ((output_df['policy_restrict_flag'] == 0) & (output_df['policy_reduce_only_flag'] == 0) & ((output_df['debt_stress_index_v1'].between(config_dict['maintain_only_dsi_min'], config_dict['maintain_only_dsi_max'])) | (output_df['active_lender_cnt_30d'] == 2))).astype(int)
    output_df['allow_decline'] = 1
    output_df['allow_restrict'] = 1
    output_df['allow_reduce'] = 1
    output_df['allow_maintain'] = (output_df['policy_restrict_flag'] == 0).astype(int)
    output_df['allow_increase_small'] = ((output_df['policy_restrict_flag'] == 0) & (output_df['policy_reduce_only_flag'] == 0)).astype(int)
    output_df['allow_increase_medium'] = ((output_df['policy_restrict_flag'] == 0) & (output_df['policy_reduce_only_flag'] == 0) & (output_df['policy_maintain_only_flag'] == 0) & (output_df['thin_file_flag'] == 0)).astype(int)
    output_df['primary_policy_reason'] = np.select(
        [
            output_df['identity_confidence_score_v1_rule'] < config_dict['identity_confidence_restrict_threshold'],
            output_df['fraud_abuse_risk_score_v1_rule'] >= config_dict['fraud_restrict_threshold'],
            output_df['debt_stress_index_v1'] >= config_dict['severe_dsi_threshold'],
            output_df['recent_disbursement_no_repayment_flag'] == 1,
            output_df['stacked_borrowing_flag_30d'] == 1,
            output_df['repayment_ratio_30d'] < config_dict['low_repayment_ratio_threshold'],
            output_df['active_lender_cnt_30d'] >= config_dict['high_active_lender_cnt_threshold'],
            output_df['exposure_to_inflow_ratio'] > config_dict['high_exposure_to_inflow_threshold'],
            output_df['thin_file_flag'] == 1
        ],
        [
            'LOW_IDENTITY_CONFIDENCE',
            'HIGH_FRAUD_ABUSE_RISK',
            'SEVERE_DSI',
            'RECENT_DISBURSEMENT_NO_REPAYMENT',
            'STACKED_BORROWING',
            'LOW_REPAYMENT_RATIO',
            'HIGH_ACTIVE_LENDER_COUNT',
            'HIGH_EXPOSURE_TO_INFLOW',
            'THIN_FILE_CONSERVATIVE_PATH'
        ],
        default='PASS'
    )
    action_cols = ['allow_decline', 'allow_restrict', 'allow_reduce', 'allow_maintain', 'allow_increase_small', 'allow_increase_medium']
    action_names = ['DECLINE', 'RESTRICT', 'REDUCE', 'MAINTAIN', 'INCREASE_SMALL', 'INCREASE_MEDIUM']
    output_df['feasible_action_set'] = output_df[action_cols].apply(lambda row_vals: ','.join([action_names[i] for i, val_num in enumerate(row_vals.tolist()) if val_num == 1]), axis=1)
    return output_df

def evaluate_tnv_actions(policy_prefilter_input_df, action_specs_input_df, config_dict):
    policy_df = policy_prefilter_input_df.copy()
    action_rows = []
    for _, subscriber_row in tqdm(policy_df.iterrows(), total=policy_df.shape[0]):
        base_capacity = max(subscriber_row['wallet_inflow_amt_30d'] * 0.35 * subscriber_row['expected_repayment_probability_v1_rule'], 0.0)
        if subscriber_row['thin_file_flag'] == 1:
            base_capacity = min(base_capacity, config_dict['thin_file_capacity_cap'])
        for _, spec_row in action_specs_input_df.iterrows():
            action_name = spec_row['action']
            allow_col_name = 'allow_' + action_name.lower()
            if subscriber_row.get(allow_col_name, 0) == 1:
                target_capacity_raw = base_capacity * spec_row['capacity_multiplier']
                immediate_lending_margin = target_capacity_raw * spec_row['base_margin_rate']
                discounted_future_transaction_margin = subscriber_row['expected_future_transaction_margin_score_v1_rule'] * subscriber_row['wallet_inflow_amt_30d'] * 0.05 * spec_row['capacity_multiplier']
                expected_credit_loss = target_capacity_raw * subscriber_row['expected_credit_loss_rate_v1_rule']
                adjusted_churn_probability = max(subscriber_row['churn_cooling_probability_v1_rule'] - spec_row['churn_relief'], 0.0)
                expected_churn_cost = adjusted_churn_probability * subscriber_row['wallet_inflow_amt_30d'] * 0.03
                expected_treatment_cost = subscriber_row['expected_treatment_cost_score_v1_rule'] * 500.0 * spec_row['treatment_multiplier']
                expected_tnv = immediate_lending_margin + discounted_future_transaction_margin - expected_credit_loss - expected_churn_cost - expected_treatment_cost
                action_rows.append({
                    'feature_dt': subscriber_row['feature_dt'],
                    'subscriber_msisdn': subscriber_row['subscriber_msisdn'],
                    'primary_policy_reason': subscriber_row['primary_policy_reason'],
                    'action': action_name,
                    'target_capacity_raw': target_capacity_raw,
                    'immediate_lending_margin': immediate_lending_margin,
                    'discounted_future_transaction_margin': discounted_future_transaction_margin,
                    'expected_credit_loss': expected_credit_loss,
                    'expected_churn_cost': expected_churn_cost,
                    'expected_treatment_cost': expected_treatment_cost,
                    'expected_tnv': expected_tnv,
                    'stability_multiplier': spec_row['stability_multiplier'],
                    'update_direction': spec_row['update_direction'],
                    'validity_days': config_dict[spec_row['validity_days_key']]
                })
    output_df = pd.DataFrame(action_rows)
    if output_df.shape[0] == 0:
        raise ValueError('No feasible actions were generated in evaluate_tnv_actions')
    return output_df

def select_final_capacity_output(tnv_actions_input_df, config_dict):
    ranked_df = tnv_actions_input_df.sort_values(['feature_dt', 'subscriber_msisdn', 'expected_tnv', 'target_capacity_raw'], ascending=[True, True, False, False]).copy()
    best_df = ranked_df.groupby(['feature_dt', 'subscriber_msisdn'], as_index=False).head(1).copy()
    best_df['selected_action'] = np.where(best_df['expected_tnv'] > 0, best_df['action'], 'DECLINE')
    best_df['target_capacity_policy_capped'] = best_df['target_capacity_raw'].clip(lower=0.0, upper=config_dict['policy_capacity_cap'])
    best_df['target_capacity_stability_adjusted'] = np.where(best_df['selected_action'] == 'DECLINE', 0.0, best_df['target_capacity_policy_capped'] * best_df['stability_multiplier'])
    best_df['CreditLimit'] = best_df['target_capacity_stability_adjusted'].round(0)
    best_df['output_version'] = config_dict['output_version']
    best_df['decision_status'] = np.where(best_df['selected_action'].isin(['DECLINE', 'RESTRICT']), 'blocked_or_restricted', 'approved_or_maintained')
    final_cols = [
        'feature_dt', 'subscriber_msisdn', 'primary_policy_reason', 'selected_action', 'decision_status',
        'expected_tnv', 'target_capacity_raw', 'target_capacity_policy_capped', 'target_capacity_stability_adjusted',
        'CreditLimit', 'update_direction', 'validity_days', 'output_version'
    ]
    return best_df[final_cols].copy()

def run_credit_v1_decision_engine(layer1_input_df, layer0_input_df, config_dict=None, action_specs_input_df=None):
    local_config = DECISION_ENGINE_CONFIG.copy() if config_dict is None else config_dict.copy()
    local_action_specs_df = ACTION_SPECS_DF.copy() if action_specs_input_df is None else action_specs_input_df.copy()
    decision_base_df = prepare_decision_base(layer1_input_df, layer0_input_df)
    policy_prefilter_df = build_policy_prefilter(decision_base_df, local_config)
    tnv_actions_df = evaluate_tnv_actions(policy_prefilter_df, local_action_specs_df, local_config)
    final_capacity_df = select_final_capacity_output(tnv_actions_df, local_config)
    return {
        'decision_base_df': decision_base_df,
        'vw7_credit_v1_policy_prefilter': policy_prefilter_df,
        'vw8_credit_v1_tnv_action_evaluation': tnv_actions_df,
        'vw9_credit_v1_final_capacity_output': final_capacity_df
    }
