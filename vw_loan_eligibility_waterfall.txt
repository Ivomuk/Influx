-- ============================================================
-- vw_loan_eligibility_waterfall
-- Applies the 6 agreed loan eligibility rules as a cumulative
-- waterfall and flags each subscriber's eligibility status.
--
-- SOURCES:
--   devdata.account_holder_dump                (subscriber registration)
--   vw1_credit_v1_normalized_events            (transaction-level events)
--   vw3_credit_v1_layer1_features              (30-day rolling features)
--   wamujap.fm_momo_loan_portfolio_cummulation (loan portfolio)
-- vw4 (scores), vw5 (reason codes), vw6 (decisions) are
-- deliberately excluded as those views are not yet verified.
--
-- DATA LIMITATION NOTES:
--   Rule 1 (MoMo age >= 180 days):
--     Source: devdata.account_holder_dump (account_type='MOBILE MONEY',
--     account_status='ACTIVE'). Derivation priority:
--       1. registration_date  (date simcard was purchased)
--       2. activation_date    (date first PIN reset was made)
--       3. first_activity_dt  (earliest vw1 transaction — fallback only)
--     registration_date and activation_date are VARCHAR YYYYMMDD ('20260401').
--     Parsed via date_parse(col, '%Y%m%d') with TRY() to handle nulls.
--   Rules 3 & 4 (transaction count / value windows):
--     Controlled by params.window_90d_start and params.analysis_end_date.
--     Currently set to a 30-day window (April 2026 data only).
--     To extend to 90 days: change window_90d_start to DATE '2026-02-01'
--     in the params CTE — no other changes needed.
--   Rule 5 (overdue > 60 days):
--     Source: wamujap.fm_momo_loan_portfolio_cummulation (loan_repaid=0).
--     Overdue measured from disbursement_date to analysis_end_date.
--     No due_date column in source; 30-day loan term assumed (confirmed).
--     Table is current and has history back to 2025. tbl_dt = 20260531 captures
--     all active unrepaid loans including those disbursed in 2025.
--   Rule 6 (cooling-off after delayed repayment):
--     Source: wamujap.fm_momo_loan_portfolio_cummulation (loan_repaid=1).
--     Late repayment proxy: days from disbursement to repayment > 30.
--     30-day loan term confirmed as default. Cooling-off = 90 days.
--     tbl_dt = 20260531 captures late repayments from early 2026 still in window.
-- ============================================================

CREATE OR REPLACE VIEW vw_loan_eligibility_waterfall AS

-- ── Analysis window parameters ────────────────────────────────
-- TO CHANGE THE ANALYSIS WINDOW: edit the four values below only.
-- window_90d_start: lower bound for Rules 3 & 4 transaction/value window.
-- Currently equals window_30d_start (30-day proxy — April 2026 data only).
-- Change window_90d_start to DATE '2026-02-01' when 90-day history is loaded.
WITH params AS (
    SELECT
        DATE '2026-05-31'  AS analysis_end_date,
        DATE '2026-05-01'  AS window_30d_start,
        DATE '2026-05-01'  AS window_90d_start,   -- still 30-day proxy (May data only)
        20260531           AS loan_snapshot_tbl_dt
),

-- ── Rule 1: subscriber registration / activation dates ────────
-- Source: devdata.account_holder_dump (MOBILE MONEY, ACTIVE accounts only).
-- registration_date and activation_date are VARCHAR YYYYMMDD (e.g. '20260401').
subscriber_reg AS (
    SELECT
        msisdn,
        TRY(date_parse(registration_date, '%Y%m%d'))  AS registration_date,
        TRY(date_parse(activation_date,   '%Y%m%d'))  AS activation_date,
        COALESCE(
            TRY(date_parse(registration_date, '%Y%m%d')),
            TRY(date_parse(activation_date,   '%Y%m%d'))
        )                                              AS derived_reg_date
    FROM devdata.account_holder_dump
    WHERE account_type   = 'MOBILE MONEY'
      AND account_status = 'ACTIVE'
      AND msisdn        != ''
),

-- ── Base: all subscribers seen in the event stream ───────────
subscriber_base AS (
    SELECT
        subscriber_msisdn,
        MIN(event_dt)            AS first_activity_dt,
        MAX(event_dt)            AS last_activity_dt,
        COUNT(DISTINCT event_dt) AS total_active_days
    FROM vw1_credit_v1_normalized_events
    GROUP BY subscriber_msisdn
),

-- ── Rule 2: MAU30 — any MoMo event in the last 30 days ───────
mau30_activity AS (
    SELECT
        v.subscriber_msisdn,
        COUNT(DISTINCT v.event_dt)                                      AS active_days_last30d,
        CASE WHEN COUNT(DISTINCT v.event_dt) >= 1 THEN 1 ELSE 0 END    AS mau30_flag
    FROM vw1_credit_v1_normalized_events v
    CROSS JOIN params p
    WHERE v.event_dt BETWEEN p.window_30d_start AND p.analysis_end_date
    GROUP BY v.subscriber_msisdn
),

-- ── Rules 3 & 4: wallet transaction count and value ──────────
-- Window controlled by params.window_90d_start → params.analysis_end_date.
-- Change window_90d_start to DATE '2026-02-01' for a proper 90-day lookback.
momo_activity AS (
    SELECT
        v.subscriber_msisdn,
        COUNT_IF(v.event_family IN ('wallet_inflow', 'wallet_outflow'))                    AS momo_txn_cnt_30d,
        SUM(CASE WHEN v.event_family = 'wallet_inflow'  THEN v.txn_value_num ELSE 0.0 END) AS momo_inflow_30d,
        SUM(CASE WHEN v.event_family = 'wallet_outflow' THEN v.txn_value_num ELSE 0.0 END) AS momo_outflow_30d
    FROM vw1_credit_v1_normalized_events v
    CROSS JOIN params p
    WHERE v.event_dt BETWEEN p.window_90d_start AND p.analysis_end_date
    GROUP BY v.subscriber_msisdn
),

-- ── Layer 1: most-recent 30-day feature snapshot per subscriber
layer1_latest AS (
    SELECT
        subscriber_msisdn,
        feature_dt,
        outstanding_exposure_amt,
        repayment_ratio_30d,
        repayment_cnt_30d,
        disbursement_cnt_30d,
        days_since_last_disbursement,
        days_since_last_repayment,
        last_disbursement_dt,
        last_repayment_dt,
        wallet_inflow_amt_30d,
        wallet_outflow_amt_30d,
        wallet_txn_cnt_30d,
        wallet_active_days_30d,
        loan_disb_amt_30d,
        loan_repaid_amt_30d,
        stacked_borrowing_flag_30d,
        repeated_borrowing_flag_30d,
        active_lender_cnt_30d,
        alt_credit_active_flag_30d
    FROM (
        SELECT *,
            ROW_NUMBER() OVER (
                PARTITION BY subscriber_msisdn
                ORDER BY feature_dt DESC
            ) AS rn
        FROM vw3_credit_v1_layer1_features
    ) t
    WHERE rn = 1
),

-- ── Rule 5: overdue loans (active, unrepaid, > 60 days old) ──
-- Snapshot date = params.analysis_end_date (not the tbl_dt snapshot date).
-- Update loan_snapshot_tbl_dt in params to 20260430 when available.
loan_overdue_signals AS (
    -- Scope: unrepaid loans disbursed within the 90-day lookback window only.
    -- Aligns with Rules 3 & 4 window and avoids pulling ancient write-off loans.
    -- A loan is overdue if it has been outstanding > 60 days within that window.
    SELECT
        msisdn,
        COUNT(DISTINCT loan_id)                                        AS overdue_loan_count,
        SUM(outstanding_loan)                                          AS total_overdue_amount,
        MAX(date_diff('day', disbursement_date, snapshot_date))        AS max_days_outstanding,
        1                                                              AS overdue_60d_flag
    FROM (
        SELECT
            lp.msisdn,
            lp.loan_id,
            lp.outstanding_loan,
            CAST(date_parse(CAST(lp.disbursement_date AS VARCHAR), '%Y%m%d') AS DATE) AS disbursement_date,
            p.analysis_end_date                                                         AS snapshot_date,
            p.window_90d_start                                                          AS lookback_start
        FROM wamujap.fm_momo_loan_portfolio_cummulation lp
        CROSS JOIN params p
        WHERE lp.tbl_dt            = p.loan_snapshot_tbl_dt
          AND lp.loan_repaid       = 0
          AND lp.disbursement_date IS NOT NULL
    ) t
    WHERE outstanding_loan > 0
      AND disbursement_date >= lookback_start               -- 90-day scope
      AND date_diff('day', disbursement_date, snapshot_date) > 60
    GROUP BY msisdn
),

-- ── Rule 6: cooling-off after late repayment ──────────────────
-- Loan term = 30 days (confirmed default — no due_date in source table).
-- "Late" = days from disbursement to repayment > 30. Cooling-off = 90 days.
-- Update loan_snapshot_tbl_dt in params to 20260430 when available.
loan_cooling_off_signals AS (
    SELECT
        msisdn,
        COUNT(DISTINCT loan_id)                           AS cooling_off_loan_count,
        MAX(repayment_date)                               AS latest_late_repayment_date,
        MAX(date_add('day', 90, repayment_date))          AS latest_cooling_off_end_date,
        SUM(loan_disbursed)                               AS cooling_off_affected_loan_value,
        1                                                 AS cooling_off_flag
    FROM (
        SELECT
            lp.msisdn,
            lp.loan_id,
            lp.loan_disbursed,
            CAST(date_parse(CAST(lp.disbursement_date AS VARCHAR), '%Y%m%d') AS DATE) AS disbursement_date,
            CAST(date_parse(CAST(lp.repayment_date    AS VARCHAR), '%Y%m%d') AS DATE) AS repayment_date,
            p.analysis_end_date                                                         AS snapshot_date
        FROM wamujap.fm_momo_loan_portfolio_cummulation lp
        CROSS JOIN params p
        WHERE lp.tbl_dt            = p.loan_snapshot_tbl_dt
          AND lp.loan_repaid       = 1
          AND lp.repayment_date    IS NOT NULL
          AND lp.disbursement_date IS NOT NULL
    ) t
    WHERE date_diff('day', disbursement_date, repayment_date) > 30
      AND snapshot_date BETWEEN repayment_date
                            AND date_add('day', 90, repayment_date)
    GROUP BY msisdn
),

-- ── Eligibility flags per subscriber ─────────────────────────
eligibility_base AS (
    SELECT
        sb.subscriber_msisdn,
        sb.first_activity_dt,
        sb.last_activity_dt,
        sb.total_active_days,

        -- Rule 1 diagnostic columns (from devdata.account_holder_dump)
        sr.derived_reg_date,
        sr.registration_date                            AS reg_registration_date,
        sr.activation_date                              AS reg_activation_date,
        date_diff('day',
            COALESCE(sr.derived_reg_date, sb.first_activity_dt),
            p.analysis_end_date
        )                                               AS momo_age_days,

        -- Rule 1: MoMo age >= 180 days
        -- Priority: registration_date → activation_date → first_activity_dt (vw1 fallback)
        CASE
            WHEN date_diff('day',
                     COALESCE(sr.derived_reg_date, sb.first_activity_dt),
                     p.analysis_end_date
                 ) >= 180
            THEN 1 ELSE 0
        END                                             AS rule1_momo_age_180d_flag,

        -- Rule 2: MAU30 — active in last 30 days
        COALESCE(m30.mau30_flag, 0)                    AS rule2_mau30_flag,
        COALESCE(m30.active_days_last30d, 0)           AS active_days_last30d,

        -- Rule 3: >= 5 MoMo transactions in 90 days (30-day proxy currently)
        CASE WHEN COALESCE(ma.momo_txn_cnt_30d, 0) >= 5
             THEN 1 ELSE 0 END                         AS rule3_min_5txn_flag,
        COALESCE(ma.momo_txn_cnt_30d, 0)              AS momo_txn_cnt_30d,

        -- Rule 4: >= UGX 10,000 inflow or outflow in 90 days (30-day proxy currently)
        CASE WHEN   COALESCE(ma.momo_inflow_30d,  0) >= 10000
               OR   COALESCE(ma.momo_outflow_30d, 0) >= 10000
             THEN 1 ELSE 0 END                         AS rule4_min_10k_value_flag,
        COALESCE(ma.momo_inflow_30d,  0)              AS momo_inflow_30d,
        COALESCE(ma.momo_outflow_30d, 0)              AS momo_outflow_30d,

        -- Rule 5: No active term loan overdue > 60 days
        -- Source: loan_overdue_signals (fm_momo_loan_portfolio_cummulation, loan_repaid=0)
        CASE WHEN lov.overdue_60d_flag = 1 THEN 0 ELSE 1 END  AS rule5_no_overdue_60d_flag,

        -- Rule 6: Not in cooling-off period
        -- Source: loan_cooling_off_signals (fm_momo_loan_portfolio_cummulation, loan_repaid=1)
        CASE WHEN lco.cooling_off_flag = 1 THEN 0 ELSE 1 END  AS rule6_no_cooling_off_flag,

        -- ── Financial features (from vw3) ──────────────────────
        COALESCE(l1.outstanding_exposure_amt,          0.0)   AS outstanding_exposure_amt,
        l1.repayment_ratio_30d,
        COALESCE(l1.repayment_cnt_30d,                 0)     AS repayment_cnt_30d,
        COALESCE(l1.disbursement_cnt_30d,              0)     AS disbursement_cnt_30d,
        COALESCE(l1.wallet_inflow_amt_30d,             0.0)   AS wallet_inflow_amt_30d,
        COALESCE(l1.wallet_outflow_amt_30d,            0.0)   AS wallet_outflow_amt_30d,
        COALESCE(l1.wallet_txn_cnt_30d,                0)     AS wallet_txn_cnt_30d,
        COALESCE(l1.wallet_active_days_30d,            0)     AS wallet_active_days_30d,
        COALESCE(l1.loan_disb_amt_30d,                 0.0)   AS loan_disb_amt_30d,
        COALESCE(l1.loan_repaid_amt_30d,               0.0)   AS loan_repaid_amt_30d,
        l1.days_since_last_disbursement,
        l1.days_since_last_repayment,
        l1.last_disbursement_dt,
        l1.last_repayment_dt,
        COALESCE(l1.stacked_borrowing_flag_30d,        0)     AS stacked_borrowing_flag_30d,
        COALESCE(l1.repeated_borrowing_flag_30d,       0)     AS repeated_borrowing_flag_30d,
        COALESCE(l1.active_lender_cnt_30d,             0)     AS active_lender_cnt_30d,
        COALESCE(l1.alt_credit_active_flag_30d,        0)     AS alt_credit_active_flag_30d,

        -- ── Loan portfolio signals (from fm_momo_loan_portfolio_cummulation)
        COALESCE(lov.overdue_loan_count,               0)     AS overdue_loan_count,
        COALESCE(lov.total_overdue_amount,             0.0)   AS total_overdue_amount,
        COALESCE(lov.max_days_outstanding,             0)     AS max_days_outstanding,
        COALESCE(lco.cooling_off_loan_count,           0)     AS cooling_off_loan_count,
        COALESCE(lco.cooling_off_affected_loan_value,  0.0)   AS cooling_off_affected_loan_value,
        lco.latest_late_repayment_date,
        lco.latest_cooling_off_end_date

    FROM subscriber_base sb
    CROSS JOIN params p
    LEFT JOIN subscriber_reg             sr  ON sb.subscriber_msisdn = sr.msisdn
    LEFT JOIN mau30_activity             m30 ON sb.subscriber_msisdn = m30.subscriber_msisdn
    LEFT JOIN momo_activity               ma ON sb.subscriber_msisdn = ma.subscriber_msisdn
    LEFT JOIN layer1_latest               l1 ON sb.subscriber_msisdn = l1.subscriber_msisdn
    LEFT JOIN loan_overdue_signals       lov ON sb.subscriber_msisdn = lov.msisdn
    LEFT JOIN loan_cooling_off_signals   lco ON sb.subscriber_msisdn = lco.msisdn
),

-- ── Waterfall: cumulative eligibility after each rule ─────────
waterfall AS (
    SELECT
        *,

        -- Baseline comparison group: subscriber had at least one
        -- loan disbursement in the current 30-day window.
        -- Used instead of "currently_eligible" (which required vw6).
        CASE WHEN disbursement_cnt_30d > 0 THEN 1 ELSE 0 END  AS currently_active_borrower_flag,

        -- Cumulative waterfall
        rule1_momo_age_180d_flag                               AS eligible_after_rule1,

        CASE WHEN rule1_momo_age_180d_flag = 1
              AND rule2_mau30_flag          = 1
             THEN 1 ELSE 0 END                                 AS eligible_after_rule2,

        CASE WHEN rule1_momo_age_180d_flag = 1
              AND rule2_mau30_flag          = 1
              AND rule3_min_5txn_flag       = 1
             THEN 1 ELSE 0 END                                 AS eligible_after_rule3,

        CASE WHEN rule1_momo_age_180d_flag = 1
              AND rule2_mau30_flag          = 1
              AND rule3_min_5txn_flag       = 1
              AND rule4_min_10k_value_flag  = 1
             THEN 1 ELSE 0 END                                 AS eligible_after_rule4,

        CASE WHEN rule1_momo_age_180d_flag = 1
              AND rule2_mau30_flag          = 1
              AND rule3_min_5txn_flag       = 1
              AND rule4_min_10k_value_flag  = 1
              AND rule5_no_overdue_60d_flag = 1
             THEN 1 ELSE 0 END                                 AS eligible_after_rule5,

        CASE WHEN rule1_momo_age_180d_flag = 1
              AND rule2_mau30_flag          = 1
              AND rule3_min_5txn_flag       = 1
              AND rule4_min_10k_value_flag  = 1
              AND rule5_no_overdue_60d_flag = 1
              AND rule6_no_cooling_off_flag = 1
             THEN 1 ELSE 0 END                                 AS proposed_eligible_flag,

        -- Revenue proxy: actual disbursement value × 10% assumed margin.
        -- No repayment probability multiplier — scores are not verified.
        loan_disb_amt_30d * 0.10                               AS estimated_monthly_revenue

    FROM eligibility_base
)

SELECT
    subscriber_msisdn,
    first_activity_dt,
    last_activity_dt,
    total_active_days,
    -- Rule 1 registration diagnostics
    derived_reg_date,
    reg_registration_date,
    reg_activation_date,
    momo_age_days,
    -- Rule flags
    rule1_momo_age_180d_flag,
    rule2_mau30_flag,
    rule3_min_5txn_flag,
    rule4_min_10k_value_flag,
    rule5_no_overdue_60d_flag,
    rule6_no_cooling_off_flag,
    -- Waterfall eligibility
    eligible_after_rule1,
    eligible_after_rule2,
    eligible_after_rule3,
    eligible_after_rule4,
    eligible_after_rule5,
    proposed_eligible_flag,
    currently_active_borrower_flag,
    -- Activity metrics
    active_days_last30d,
    momo_txn_cnt_30d,
    momo_inflow_30d,
    momo_outflow_30d,
    -- Financial features (vw3)
    outstanding_exposure_amt,
    repayment_ratio_30d,
    repayment_cnt_30d,
    disbursement_cnt_30d,
    wallet_inflow_amt_30d,
    wallet_outflow_amt_30d,
    wallet_txn_cnt_30d,
    wallet_active_days_30d,
    loan_disb_amt_30d,
    loan_repaid_amt_30d,
    days_since_last_disbursement,
    days_since_last_repayment,
    last_disbursement_dt,
    last_repayment_dt,
    -- Risk behaviour flags (vw3)
    stacked_borrowing_flag_30d,
    repeated_borrowing_flag_30d,
    active_lender_cnt_30d,
    alt_credit_active_flag_30d,
    -- Loan portfolio signals (fm_momo_loan_portfolio_cummulation)
    overdue_loan_count,
    total_overdue_amount,
    max_days_outstanding,
    cooling_off_loan_count,
    cooling_off_affected_loan_value,
    latest_late_repayment_date,
    latest_cooling_off_end_date,
    -- Revenue estimate
    estimated_monthly_revenue

FROM waterfall
;
