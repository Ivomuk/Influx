insert into {TARGET_MART_TABLE}

with

-- ─────────────────────────────────────────────────────────────────────────────
-- Rules 3 & 4: 90-day transaction activity from fin_log_cfw
-- Outbound = subscriber as sender; Inbound = subscriber as receiver.
-- Only subscriber profiles are counted (excludes agents, merchants, etc.).
-- ─────────────────────────────────────────────────────────────────────────────
raw_txns as (
    select
        instruct_hdr_fid,
        instruct_from_fro_msisdn    as msisdn,
        instruct_amount,
        'OUT'                       as direction
    from feeds.fin_log_cfw
    where tbl_dt between cast(date_format(date_add('day', -89, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
                     and {RUN_DATE_KEY}
      and status_code = 'EXECUTED'
      and hdr_type = 'RESERVATION'
      and instruct_from_fro_msisdn is not null
      and instruct_from_fro_user_prf in (
            select profile_name from analytics.mobile_money_profiles where subscriber_flag = true
          )

    union all

    select
        instruct_hdr_fid,
        instruct_to_fro_msisdn      as msisdn,
        instruct_amount,
        'IN'                        as direction
    from feeds.fin_log_cfw
    where tbl_dt between cast(date_format(date_add('day', -89, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
                     and {RUN_DATE_KEY}
      and status_code = 'EXECUTED'
      and hdr_type = 'RESERVATION'
      and instruct_to_fro_msisdn is not null
      and instruct_to_fro_user_prf in (
            select profile_name from analytics.mobile_money_profiles where subscriber_flag = true
          )
),

sub_activity as (
    select
        msisdn,
        count(distinct instruct_hdr_fid)                                    as txn_count,
        sum(case when direction = 'IN'  then instruct_amount else 0 end)    as inflow_amt,
        sum(case when direction = 'OUT' then instruct_amount else 0 end)    as outflow_amt
    from raw_txns
    group by msisdn
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 2: Customer must be MAU30 on the run date.
-- ─────────────────────────────────────────────────────────────────────────────
mau_30 as (
    select distinct msisdn
    from analytics.momo_rge_base_subscribers_v9
    where date_key = {RUN_DATE_KEY}
      and au_30_day_active_flag = 1
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 1: Age on MoMo >= 180 days.
-- A subscriber who was active 179 days ago has been on the platform >= 180 days.
-- ─────────────────────────────────────────────────────────────────────────────
mau_180 as (
    select distinct msisdn
    from analytics.momo_rge_base_subscribers_v9
    where date_key = cast(date_format(date_add('day', -179, date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')), '%Y%m%d') as int)
      and au_1_day_active_flag = 1
   
  -- Change source to devdata.account_holder_dump
  -- select * from devdata.account_holder_dump where tbl_dt = 20260930 limit 10
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Derisk exclusion: MTNU HV Gold Subscriber and MTNU Agent Derisk Class
-- profiles are systematically excluded from lending partner distribution.
-- ─────────────────────────────────────────────────────────────────────────────
derisk_exclusions as (
    select distinct msisdn
    from devdata.account_holder_dump
    where tbl_dt = {RUN_DATE_KEY}
      and profile in ('MTNU Agent Derisk Class', 'MTNU HV Gold Subscriber')
   
 -- Double check if this rule is VALID
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 5: No active term loan overdue by more than 60 days.
-- Active = not repaid, outstanding balance > 0, disbursed > 60 days ago.
-- ─────────────────────────────────────────────────────────────────────────────
active_overdue_60d as (
    select distinct msisdn
    from wamujap.fm_momo_loan_portfolio_cummulation
    where tbl_dt = {RUN_DATE_KEY}
      and loan_repaid = 0
      and outstanding_loan > 0
      and disbursement_date is not null
      and date_diff(
            'day',
            date_parse(cast(disbursement_date as varchar), '%Y%m%d'),
            date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')
          ) > 60
    
 -- Change source to analytics.momo_loan_book_tracker_loan_state_daily
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Rule 6: 90-day cooling-off after a late repayment.
-- A loan is "late" when repayment took > 30 days from disbursement.
-- Cooling-off runs 90 days from the actual repayment date.
-- ─────────────────────────────────────────────────────────────────────────────
cooling_off as (
    select distinct msisdn
    from (
        select
            msisdn,
            date_add(
                'day',
                90,
                date_parse(cast(repayment_date as varchar), '%Y%m%d')
            ) as cooling_off_end_dt
        from wamujap.fm_momo_loan_portfolio_cummulation
        where tbl_dt = {RUN_DATE_KEY}
          and loan_repaid = 1
          and repayment_date is not null
          and disbursement_date is not null
          and date_diff(
                'day',
                date_parse(cast(disbursement_date as varchar), '%Y%m%d'),
                date_parse(cast(repayment_date as varchar), '%Y%m%d')
              ) > 30
    ) late_repaid
    where cooling_off_end_dt >= date_parse(cast({RUN_DATE_KEY} as varchar), '%Y%m%d')
 
 -- Change source to analytics.momo_loan_book_tracker_loan_state_daily
),

-- ─────────────────────────────────────────────────────────────────────────────
-- Final eligibility gate: all 6 rules must pass.
-- Rules 1 & 2 → inner joins on mau_180 and mau_30.
-- Rules 3 & 4 → txn_count >= 5 AND (inflows + outflows) >= UGX 10,000.
-- Rules 5 & 6 → NOT IN overdue / cooling-off sets.
-- Derisk profiles are also excluded.
-- ─────────────────────────────────────────────────────────────────────────────
eligible as (
    select
        m30.msisdn,
        coalesce(a.txn_count,   0)  as txn_count_90d,
        coalesce(a.inflow_amt,  0)  as inflow_amt_90d,
        coalesce(a.outflow_amt, 0)  as outflow_amt_90d
    from mau_30 m30
    inner join mau_180 m180
        on m30.msisdn = m180.msisdn
    inner join sub_activity a
        on m30.msisdn = a.msisdn
    where a.txn_count >= 5
      and (a.inflow_amt + a.outflow_amt) >= 10000
      and m30.msisdn not in (select msisdn from derisk_exclusions)
      and m30.msisdn not in (select msisdn from active_overdue_60d)
      and m30.msisdn not in (select msisdn from cooling_off)
)

select
    msisdn,
    txn_count_90d,
    inflow_amt_90d,
    outflow_amt_90d,
    cast(1 as tinyint)      as eligible_flag,
    current_timestamp       as inserted_dt,
    {RUN_DATE_KEY}          as tbl_dt
from eligible
